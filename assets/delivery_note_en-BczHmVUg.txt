}

function hideCols(){

 var tbl = document.getElementById('listing_table');
    var rows = tbl.getElementsByTagName('tr');

    var label_idx = document.getElementById('label_field1').cellIndex;
    var qty_idx = document.getElementById('label_quantity').cellIndex;

    for(var row=0; row<rows.length; row++){

        var cels = rows[row].cells
        var label = rows[row].cells[label_idx]
        var qty = rows[row].cells[qty_idx]

        for(var i=0; i<cels.length; i++){
cels[i].style.borderRight='1px solid black';
            if (!(cels[i].cellIndex == label_idx || cels[i].cellIndex == qty_idx)){
                cels[i].classList.add('hidden');

    }}}

}

function newColClass(){
   var allRows = document.getElementById('listing_table').rows
       for(var i=0; i<allRows.length;i++){
             for(var x =0; x< allRows[i].cells.length; x++){
                         allRows[i].cells[x].className = ""
                           allRows[i].cells[x].classList.add('col-'+(x+1))
                    }
           }
  }

   var mainDate = document.getElementById('main_date')

      mainNo.classList.add('hidden')
     }

 if(customDate.innerText.length != 0){
      mainDate.classList.add('hidden')
     }

}
